name: Release CI

on:
  push:
    tags:
      - v*

permissions:
  contents: write

jobs:
  generate-changelog:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
          ref: gokul-dev
      - uses: actions/setup-node@v4
        with:
          node-version: 18.x
      - run: npx changelogithub
        env:
          GITHUB_TOKEN: ${{secrets.GH_TOKEN}}

  build-and-release-windows:
    runs-on: windows-latest

    steps:
      - uses: actions/checkout@v4
        with:
          ref: gokul-dev
      - uses: subosito/flutter-action@v2.12.0
        with:
          flutter-version: 3.27.1
          channel: stable
      - name: Install project dependencies
        run: flutter pub get
      - name: Build artifacts
        run: flutter build windows --release -v
      - name: Build Installer
        run: iex ("& {0} {1} {2}" -f '"C:\Program Files (x86)\Inno Setup 6\iscc.exe"',"/DMyAppVersion='${{github.ref_name}}'", ".\inno_setup.iss")
      - name: Archive Release
        uses: thedoctor0/zip-release@master
        with:
          type: "zip"
          filename: EposMachine-${{github.ref_name}}-windows.zip
          directory: build/windows/x64/runner/Release
      - name: Rename Release Directory Name to EposMachine
        run: |
          mkdir temp
          mv build/windows/x64/runner/Release/EposMachine-${{github.ref_name}}-windows.zip temp/EposMachine-${{github.ref_name}}-windows.zip
          mv build/windows/x64/runner/Release build/windows/x64/runner/EposMachine

      # Updated artifact upload step
      - name: Upload Artifact
        uses: actions/upload-artifact@v4
        with:
          path: "build/windows/x64/runner/EposMachine"
          name: EposMachine-windows-${{github.ref_name}}.zip

      - name: Create Release
        uses: softprops/action-gh-release@v2
        with:
          files: |
            temp/EposMachine-${{github.ref_name}}-windows.zip
            Output/*.exe
          draft: false
          prerelease: false
          generate_release_notes: true
